  5.1 Listar e excluir (peso 2) | 5.2 Busca com termo (peso 3)
  5.3 Inserir (peso 2)          | 5.5 Editar (peso 3)

Ferramentas do DOM:
- document.getElementById('id')  -> pega o elemento
- .value                         -> lê/escreve o que está no input
- .textContent                   -> escreve texto
- .innerHTML                     -> escreve HTML (para montar a tabela)
- array.forEach(item => ...)     -> percorre o que veio da API
- array.find(item => ...)        -> acha um item (usado no editar)

Regra de ouro da lista: limpe antes (innerHTML = '') e some com += dentro do forEach.

// ### Listar (com busca opcional)

let barbeiros = []

async function carregar(busca) {
    const caminho = busca ? `/barbeiros?busca=${encodeURIComponent(busca)}` : '/barbeiros'
    barbeiros = await api(caminho)

    const lista = document.getElementById('lista')
    lista.innerHTML = ''

    barbeiros.forEach(barbeiro => {
        lista.innerHTML += `
            <tr>
                <td>${barbeiro.id}</td>
                <td>${barbeiro.nome}</td>
                <td>${barbeiro.email}</td>
                <td>${barbeiro.telefone || ''}</td>
                <td>
                    <button onclick="editar(${barbeiro.id})">Editar</button>
                    <button onclick="excluir(${barbeiro.id})">Excluir</button>
                </td>
            </tr>
        `
    })
}


// ### Buscar: inserir o termo, confirmar e a listagem atualiza

function buscar(event) {
    event.preventDefault()
    carregar(document.getElementById('termo').value)
}

function limparBusca() {
    document.getElementById('termo').value = ''
    carregar()
}


// ### Salvar: o MESMO formulário cadastra (POST) e edita (PUT)

async function salvar(event) {
    event.preventDefault()

    const id = document.getElementById('id').value
    const barbeiro = {
        nome: document.getElementById('nome').value,
        email: document.getElementById('email').value,
        cpf: document.getElementById('cpf').value,
        senha: document.getElementById('senha').value,
        telefone: document.getElementById('telefone').value
    }

    let dados
    if (id) {
        dados = await api(`/barbeiros/${id}`, 'PUT', barbeiro)
    } else {
        dados = await api('/barbeiros', 'POST', barbeiro)
    }

    alert(dados.message)
    limparFormulario()
    carregar()
}


// ### Editar: joga os dados do barbeiro no formulário

function editar(id) {
    const barbeiro = barbeiros.find(b => b.id === id)

    document.getElementById('id').value = barbeiro.id
    document.getElementById('nome').value = barbeiro.nome
    document.getElementById('email').value = barbeiro.email
    document.getElementById('telefone').value = barbeiro.telefone || ''
    document.getElementById('cpf').placeholder = 'Novo CPF (opcional)'
    document.getElementById('senha').placeholder = 'Nova senha (opcional)'

    document.getElementById('tituloForm').textContent = 'Editar barbeiro'
    document.getElementById('botaoSalvar').textContent = 'Salvar alterações'
}

function limparFormulario() {
    document.getElementById('formBarbeiro').reset()
    document.getElementById('id').value = ''
    document.getElementById('cpf').placeholder = 'CPF'
    document.getElementById('senha').placeholder = 'Senha'
    document.getElementById('tituloForm').textContent = 'Cadastrar barbeiro'
    document.getElementById('botaoSalvar').textContent = 'Cadastrar'
}


// ### Excluir

async function excluir(id) {
    if (!confirm('Deseja excluir este barbeiro?')) return

    const dados = await api(`/barbeiros/${id}`, 'DELETE')
    alert(dados.message)
    carregar()
}
