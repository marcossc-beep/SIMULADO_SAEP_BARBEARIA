// ### Introdução a Banco de Dados (saep_db.sql) - ENTREGA 3

Um banco de dados relacional é como uma planilha gigante organizada em tabelas:
- Tabela: uma aba da planilha (barbeiro, cliente, agendamento).
- Colunas: o tipo de informação (nome, email, cpf, data_agendamento).
- Linhas (registros): os dados reais ("Rafael Navalha", "rafael@barbearia.com").

Critérios que valem ponto nessa entrega:
  2.1 Criou as tabelas respeitando a chave estrangeira de cada relacionamento (peso 2)
  2.2 Inseriu pelo menos TRÊS registros em CADA tabela (peso 1)


// ### Passo 0: criar o banco no pgAdmin

CREATE DATABASE saep_db;

// Depois clique no banco saep_db > Query Tool (o script roda DENTRO dele).


// ### Tipos de dados e regras

- SERIAL: número que aumenta sozinho (1, 2, 3...). Usado no id.
- VARCHAR(100): texto com limite. Nome, e-mail, telefone.
- VARCHAR(255): usamos para senha e CPF porque o HASH do bcrypt tem 60 caracteres.
- TIMESTAMP: data e hora juntas (2026-11-18 14:30:00).
- INT: número inteiro. Toda chave estrangeira é INT.
- PRIMARY KEY: identificador único da linha.
- NOT NULL: obrigatório.
- UNIQUE: não deixa repetir (dois barbeiros com o mesmo e-mail).
- REFERENCES tabela(id): cria a CHAVE ESTRANGEIRA.
- ON DELETE CASCADE: se apagar o barbeiro, apaga junto os clientes e agendamentos dele
  (sem isso o DELETE do barbeiro dá erro de chave estrangeira).


// ### Extensão pgcrypto (para inserir senha e CPF já criptografados no script)

// O Postgres tem a função crypt() que gera hash BCRYPT, o mesmo formato que o
// bcrypt do Node entende. Assim o login funciona com os usuários do script.

CREATE EXTENSION IF NOT EXISTS pgcrypto;

// crypt('123', gen_salt('bf'))  ->  $2a$06$....   (bf = blowfish = bcrypt)


// ### ORDEM DE CRIAÇÃO: primeiro quem NÃO depende de ninguém

CREATE TABLE usuario (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    senha VARCHAR(255) NOT NULL
);

CREATE TABLE barbeiro (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    cpf VARCHAR(255) NOT NULL,
    senha VARCHAR(255) NOT NULL,
    telefone VARCHAR(20)
);

-- cliente depende de barbeiro (1 barbeiro tem N clientes)
CREATE TABLE cliente (
    id SERIAL PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    telefone VARCHAR(20),
    estilo_corte VARCHAR(100),
    barbeiro_id INT NOT NULL REFERENCES barbeiro(id) ON DELETE CASCADE
);

-- agendamento depende de barbeiro E de cliente
CREATE TABLE agendamento (
    id SERIAL PRIMARY KEY,
    data_agendamento TIMESTAMP NOT NULL,
    servico VARCHAR(100) NOT NULL,
    barbeiro_id INT NOT NULL REFERENCES barbeiro(id) ON DELETE CASCADE,
    cliente_id INT NOT NULL REFERENCES cliente(id) ON DELETE CASCADE
);

// ### Consultas que a API vai usar

-- Busca por termo (ILIKE ignora maiúsculas/minúsculas, % = qualquer coisa antes/depois)
SELECT id, nome, email, telefone FROM barbeiro
WHERE nome ILIKE '%raf%' OR email ILIKE '%raf%'
ORDER BY id;

-- Cliente com o nome do barbeiro (JOIN de 2 tabelas)
SELECT cliente.*, barbeiro.nome AS barbeiro_nome
FROM cliente
INNER JOIN barbeiro ON cliente.barbeiro_id = barbeiro.id;

-- Agendamentos ordenados por data com dados do barbeiro e do cliente (JOIN de 3 tabelas)
SELECT agendamento.id, agendamento.data_agendamento, agendamento.servico,
       barbeiro.nome AS barbeiro_nome, barbeiro.email AS barbeiro_email, barbeiro.telefone AS barbeiro_telefone,
       cliente.nome AS cliente_nome, cliente.telefone AS cliente_telefone, cliente.estilo_corte AS cliente_estilo_corte
FROM agendamento
INNER JOIN barbeiro ON agendamento.barbeiro_id = barbeiro.id
INNER JOIN cliente ON agendamento.cliente_id = cliente.id
ORDER BY agendamento.data_agendamento ASC;

-- Atualizar só a data (SEMPRE com WHERE!)
UPDATE agendamento SET data_agendamento = '2026-11-17 08:30:00' WHERE id = 1;

- "function gen_salt does not exist": esqueceu o CREATE EXTENSION pgcrypto.
- Rodou o script 2 vezes: apague as tabelas antes (DROP TABLE agendamento, cliente, barbeiro, usuario;).
