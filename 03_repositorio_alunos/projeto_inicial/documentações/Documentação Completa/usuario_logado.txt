// ### localStorage: a "memória" do navegador

localStorage.setItem('chave', valor)   -> salva (só texto)
localStorage.getItem('chave')          -> lê (ou null)
localStorage.removeItem('chave')       -> apaga (usado no Sair)
JSON.stringify(objeto) / JSON.parse(texto) -> objeto <-> texto

Depois do login guardamos 3 coisas: token, expira e usuario.


// ### Backend: middleware que protege as rotas

async function autenticar(request, reply) {
    try {
        await request.jwtVerify()
    } catch (erro) {
        return reply.status(401).send({ message: 'Sessão expirada ou inválida. Faça login novamente.' })
    }
}

// Uso: servidor.get('/barbeiros', { preHandler: autenticar }, async (request, reply) => { ... })


// ### Front: sessao.js (incluído em TODAS as páginas com <script src="sessao.js">)

const API = 'http://localhost:3000'

function pegarUsuario() {
    return JSON.parse(localStorage.getItem('usuario'))
}

function sessaoValida() {
    const token = localStorage.getItem('token')
    const expira = Number(localStorage.getItem('expira'))
    return token && Date.now() < expira
}

function sair(mensagem) {
    localStorage.removeItem('token')
    localStorage.removeItem('usuario')
    localStorage.removeItem('expira')
    if (mensagem) alert(mensagem)
    window.location.href = 'index.html'
}

function protegerPagina() {
    if (!sessaoValida()) {
        sair('Sessão expirada. Faça login novamente.')
    }
}

// fetch que já manda o token e trata a sessão expirada
async function api(caminho, metodo = 'GET', corpo) {
    const opcoes = {
        method: metodo,
        headers: { Authorization: 'Bearer ' + localStorage.getItem('token') }
    }

    if (corpo) {
        opcoes.headers['Content-Type'] = 'application/json'
        opcoes.body = JSON.stringify(corpo)
    }

    const resposta = await fetch(API + caminho, opcoes)
    const dados = await resposta.json()

    if (resposta.status === 401) {
        sair(dados.message)
    }

    return dados
}

function formatarData(data) {
    return new Date(data).toLocaleString('pt-BR', { dateStyle: 'short', timeStyle: 'short' })
}


// ### index.html (login)

<form id="formLogin" onsubmit="fazerLogin(event)">
    <input type="email" id="email" placeholder="E-mail">
    <input type="password" id="senha" placeholder="Senha">
    <button type="submit">Entrar</button>
    <p id="mensagem" class="erro"></p>
</form>

<script src="sessao.js"></script>
<script>
    if (sessaoValida()) {
        window.location.href = 'principal.html'
    }

    async function fazerLogin(event) {
        event.preventDefault()

        const email = document.getElementById('email').value
        const senha = document.getElementById('senha').value
        const mensagem = document.getElementById('mensagem')

        // Validação no FRONT (campos vazios)
        if (!email || !senha) {
            mensagem.textContent = 'Preencha e-mail e senha!'
            return
        }

        const resposta = await fetch(`${API}/login`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ email, senha })
        })
        const dados = await resposta.json()

        // Validação do BACK (usuário ou senha inválidos)
        if (!resposta.ok) {
            mensagem.textContent = dados.message
            return
        }

        localStorage.setItem('token', dados.token)
        localStorage.setItem('expira', dados.expira)
        localStorage.setItem('usuario', JSON.stringify(dados.usuario))

        window.location.href = 'principal.html'
    }
</script>
